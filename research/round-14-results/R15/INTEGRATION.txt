  git fetch origin claude/r14-R15
  for p in P-R01 P-R09 P-R07 P-R04 P-R02 P-R06 P-R05; do
      git show origin/claude/r14-R15:research/round-14-results/R15/patches/$p.diff | git apply --check - && \
      git show origin/claude/r14-R15:research/round-14-results/R15/patches/$p.diff | git apply -
  done
  # проверка: тесты R15 должны стать XPASS (strict -> «failed» с пометкой XPASS) — значит, исправлено
  git checkout origin/claude/r14-R15 -- tests/civic/R15

  # затем R01 переносит ID в FIXED (tests/civic/R15/r15_common.py) и снимает xfail — или пишет R15, сделаю я.
