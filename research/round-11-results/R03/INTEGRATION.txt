    map: window.GOVTECH.map,                           // the one MapLibre instance
    api,                                               // R01 api.request(method, path, body) -> Promise<data>
    onSelect: (obj, meta) => {...},                    // obj = {id,title,kind,status,geometry,...} | null
    onFeedback: ({objectId, geometry}) => window.CivicFeedback.mount({root, api, objectId, geometry}),
  });
  ...
  civicMap.destroy();   // on leaving the civic mode
